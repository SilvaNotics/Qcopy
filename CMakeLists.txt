cmake_minimum_required(VERSION 3.24)
project(Qcopy VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Qcopy requires an x64 MSVC Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /utf-8)
add_library(qcopy_engine STATIC src/engine.cpp src/win.cpp src/transfer.cpp)
target_include_directories(qcopy_engine PUBLIC src)
target_link_libraries(qcopy_engine PUBLIC bcrypt)
add_executable(qcopy src/cli.cpp)
target_link_libraries(qcopy PRIVATE qcopy_engine)
add_executable(QcopyDesktop WIN32 src/desktop.cpp)
target_link_libraries(QcopyDesktop PRIVATE qcopy_engine comctl32 ole32 shell32 uuid gdi32)
foreach(app qcopy QcopyDesktop)
  target_link_options(${app} PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/qcopy.manifest)
endforeach()
install(TARGETS qcopy QcopyDesktop RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(qcopy_fault_test tests/faults.cpp src/cli.cpp src/engine.cpp src/transfer.cpp src/win.cpp)
  target_include_directories(qcopy_fault_test PRIVATE src)
  target_compile_definitions(qcopy_fault_test PRIVATE QCOPY_INSTRUMENTED)
  target_link_libraries(qcopy_fault_test PRIVATE bcrypt)
  add_executable(qcopy_path_test tests/paths.cpp)
  target_link_libraries(qcopy_path_test PRIVATE qcopy_engine)
  add_test(NAME unicode_long_path COMMAND qcopy_path_test $<TARGET_FILE_DIR:qcopy>)
  add_executable(qcopy_shell_bench tests/shell_copy.cpp)
  target_link_libraries(qcopy_shell_bench PRIVATE ole32 shell32 uuid)
  add_test(NAME cli COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.ps1 -Exe $<TARGET_FILE:qcopy>)
  add_test(NAME volumes COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/volumes.ps1 -Exe $<TARGET_FILE:qcopy>)
  add_test(NAME fault_transactions COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/faults.ps1 -Exe $<TARGET_FILE:qcopy_fault_test>)
  set_tests_properties(volumes fault_transactions PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME desktop COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/desktop.ps1 -Exe $<TARGET_FILE:QcopyDesktop>)
endif()
